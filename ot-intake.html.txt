      inner = '<div class="card"><h2>'+esc(s.title)+'</h2>'+(s.sub?'<p class="sub">'+esc(s.sub)+'</p>':'')+s.html+navButtons()+'</div>';
    }
    $("stage").innerHTML = inner;
    var root=$("stage");
    bindInputs(root); bindMulti(root); bindUploaders(root);
    wireSpecial(root);
    setProgress();
    window.scrollTo({top:0,behavior:"smooth"});
  }
  function navButtons(){
    var last = step===STEPS.length-2; // Review step
    return '<div class="btns">'+(step>1?'<button class="btn ghost" data-prev>Previous</button>':'')+
      '<button class="btn primary" data-'+(last?"submit":"next")+'>'+(last?"Submit Intake":"Continue")+'</button></div>';
  }

  function wireSpecial(root){
    root.querySelectorAll("[data-next]").forEach(function(b){b.addEventListener("click",function(){ save(); step++; render(); });});
    root.querySelectorAll("[data-prev]").forEach(function(b){b.addEventListener("click",function(){ step--; render(); });});
    root.querySelectorAll("[data-goto]").forEach(function(b){b.addEventListener("click",function(e){e.preventDefault();step=parseInt(b.getAttribute("data-goto"),10);render();});});
    root.querySelectorAll("[data-submit]").forEach(function(b){b.addEventListener("click",submit);});
    // caregiver primary fields
    root.querySelectorAll("[data-cg]").forEach(function(el){el.addEventListener("input",function(){A.caregivers[0]=A.caregivers[0]||{};A.caregivers[0][el.getAttribute("data-cg")]=el.value;});});
    // second caregiver add
    var addcg=root.querySelector("[data-addcg]");
    if(addcg)addcg.addEventListener("click",function(){ if($("cg2").innerHTML)return; A.caregivers[1]=A.caregivers[1]||{}; var c=A.caregivers[1];
      $("cg2").innerHTML='<div style="margin-top:16px;padding-top:12px;border-top:1px dashed var(--line)"><label class="f">Second Caregiver Name</label><input type="text" data-cg2="name" value="'+esc(c.name||"")+'"/><div class="row"><div class="col"><label class="f">Phone</label><input type="tel" data-cg2="phone" value="'+esc(c.phone||"")+'"/></div><div class="col"><label class="f">Email</label><input type="email" data-cg2="email" value="'+esc(c.email||"")+'"/></div></div></div>';
      $("cg2").querySelectorAll("[data-cg2]").forEach(function(el){el.addEventListener("input",function(){A.caregivers[1][el.getAttribute("data-cg2")]=el.value;});});
    });
    // insurance fields
    root.querySelectorAll("[data-ins]").forEach(function(el){el.addEventListener("input",function(){A.insurance=A.insurance||{};A.insurance[el.getAttribute("data-ins")]=el.value;});});
    var addsec=root.querySelector("[data-addsec]");
    if(addsec)addsec.addEventListener("click",function(){ if($("sec2").innerHTML)return; A.secondary=A.secondary||{};
      $("sec2").innerHTML='<div style="margin-top:12px;padding-top:12px;border-top:1px dashed var(--line)"><label class="f">Secondary Insurance Provider</label><input type="text" data-sec="provider" value="'+esc(A.secondary.provider||"")+'"/><label class="f">Secondary Member ID</label><input type="text" data-sec="member_id" value="'+esc(A.secondary.member_id||"")+'"/><label class="f">Secondary Card</label><div class="row"><div class="col">'+uploader("secondary_front","Front","Optional")+'</div><div class="col">'+uploader("secondary_back","Back","Optional")+'</div></div></div>';
      $("sec2").querySelectorAll("[data-sec]").forEach(function(el){el.addEventListener("input",function(){A.secondary[el.getAttribute("data-sec")]=el.value;});});
      bindUploaders($("sec2"));
    });
    // previous OT yes/no
    root.querySelectorAll("[data-prev]").forEach(function(el){ if(el.hasAttribute("data-prev")&&el.classList.contains("opt")){
      el.addEventListener("click",function(){
        A.previous_ot_nv = el.getAttribute("data-prev")==="yes";
        root.querySelectorAll(".opt[data-prev]").forEach(function(x){x.classList.remove("sel");x.querySelector(".tick").textContent="";});
        el.classList.add("sel");el.querySelector(".tick").textContent="✓";
        renderDischarge();
      });
    }});
    renderDischarge();
    // flexibility
    root.querySelectorAll("[data-flex]").forEach(function(el){el.addEventListener("click",function(){
      A.flexibility=el.getAttribute("data-flex");
      root.querySelectorAll("[data-flex]").forEach(function(x){x.classList.remove("sel");x.querySelector(".tick").textContent="";});
      el.classList.add("sel");el.querySelector(".tick").textContent="✓";
    });});
    // availability slots
    root.querySelectorAll(".slot").forEach(function(b){b.addEventListener("click",function(){
      var d=b.getAttribute("data-day"),p=b.getAttribute("data-part");
      A.availability[d]=A.availability[d]||{}; A.availability[d][p]=!A.availability[d][p];
      b.classList.toggle("sel",A.availability[d][p]);
    });});
    // interpreter language (barriers)
    renderLang(root);
    root.querySelectorAll('[data-multi="barriers"]').forEach(function(el){el.addEventListener("click",function(){setTimeout(function(){renderLang(root);},0);});});
  }
  function renderDischarge(){
    var box=$("dischargebox"); if(!box)return;
    if(A.previous_ot_nv===true){
      box.innerHTML='<div style="margin-top:12px;background:#fff7e6;border:1px solid #f0e2bf;border-radius:12px;padding:12px 14px;font-size:13.5px;color:#7a5a12">To transition OT services, we may need a discharge letter from your child’s previous OT provider.</div>'+
        '<div style="margin-top:10px">'+uploader("discharge_letter","Upload Previous OT Discharge Letter","Photo or PDF")+'</div>'+
        '<div class="opt'+(A.no_discharge_yet?" sel":"")+'" data-nodischarge style="margin-top:10px"><span class="tick">'+(A.no_discharge_yet?"✓":"")+'</span>I don’t have this yet</div>';
      bindUploaders(box);
      var nd=box.querySelector("[data-nodischarge]");
      nd.addEventListener("click",function(){A.no_discharge_yet=!A.no_discharge_yet;nd.classList.toggle("sel",A.no_discharge_yet);nd.querySelector(".tick").textContent=A.no_discharge_yet?"✓":"";});
    } else { box.innerHTML=""; }
  }
  function renderLang(root){
    var box=root.querySelector("#langbox"); if(!box)return;
    var need=(A.barriers||[]).indexOf("Language interpreter needed")>=0;
    if(need && !box.innerHTML){ box.innerHTML='<label class="f">Preferred language</label><input type="text" data-k="interpreter_language" value="'+esc(A.interpreter_language||"")+'"/>'; bindInputs(box); }
    else if(!need){ box.innerHTML=""; }
  }

  function submit(){
    var consent=$("consent"); var err=$("reviewErr");
    if(consent && !consent.checked){ err.textContent="Please check the box to confirm and authorize."; return; }
    if(!A.signed_name || !A.signed_name.trim()){ err.textContent="Please type your full name to sign."; return; }
    var btn=document.querySelector("[data-submit]"); if(btn){btn.disabled=true;btn.textContent="Submitting…";}
    apiReq(API+"/"+token+"/submit",{method:"POST",body:{data:A,signed_name:A.signed_name}})
      .then(function(){ step=STEPS.length-1; render(); })
      .catch(function(e){ err.textContent=e.message||"Could not submit. Please try again."; if(btn){btn.disabled=false;btn.textContent="Submit Intake";} });
  }

  // ---------- boot ----------
  function boot(){
    var url=new URL(location.href);
    var existing=url.searchParams.get("token");
    if(existing){
      token=existing;
      apiReq(API+"/"+token).then(function(d){ if(d.submitted){step=STEPS.length-1;render();return;} A=Object.assign(A,d.data||{}); if(!A.caregivers||!A.caregivers.length)A.caregivers=[{}]; render(); })
        .catch(function(){ startFresh(); });
    } else { startFresh(); }
  }
  function startFresh(){
    apiReq(API+"/start",{method:"POST"}).then(function(d){ token=d.token; var u=new URL(location.href); u.searchParams.set("token",token); history.replaceState(null,"",u.toString()); render(); })
      .catch(function(){ $("stage").innerHTML='<div class="card"><h2>We’re sorry</h2><p class="sub">We couldn’t start the intake right now. Please refresh, or contact our team and we’ll help you get started.</p></div>'; });
  }
  boot();
})();
